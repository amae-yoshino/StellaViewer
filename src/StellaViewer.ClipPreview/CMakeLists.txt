cmake_minimum_required(VERSION 3.24)
cmake_policy(SET CMP0091 NEW)
project(StellaViewerClipPreview LANGUAGES C CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_library(StellaViewer.FileOperations SHARED "${CMAKE_CURRENT_SOURCE_DIR}/../StellaViewer.FileOperations/shell.cpp")
target_compile_features(StellaViewer.FileOperations PRIVATE cxx_std_17)
target_link_libraries(StellaViewer.FileOperations PRIVATE shell32 ole32 uuid)
target_compile_options(StellaViewer.FileOperations PRIVATE /utf-8 /W4 /WX)
include(FetchContent)
FetchContent_Declare(sqlite URL https://sqlite.org/2026/sqlite-amalgamation-3540000.zip
    URL_HASH SHA256=68e913b0fe8ec6b4e5f391c594a28872884e80d6cd042a1e50c957be46b5aa4d DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(sqlite)
add_library(clip_sqlite STATIC "${sqlite_SOURCE_DIR}/sqlite3.c")
target_include_directories(clip_sqlite PUBLIC "${sqlite_SOURCE_DIR}")
target_compile_definitions(clip_sqlite PRIVATE SQLITE_OMIT_LOAD_EXTENSION=1 SQLITE_THREADSAFE=0 SQLITE_DQS=0 SQLITE_DEFAULT_MEMSTATUS=1 SQLITE_TEMP_STORE=3)
add_executable(StellaViewer.ClipPreview main.cpp)
target_compile_features(StellaViewer.ClipPreview PRIVATE cxx_std_17)
target_link_libraries(StellaViewer.ClipPreview PRIVATE clip_sqlite)
add_executable(StellaViewer.ClipFixtures "${CMAKE_CURRENT_SOURCE_DIR}/../../tests/StellaViewer.ClipFixtures/main.cpp")
target_compile_features(StellaViewer.ClipFixtures PRIVATE cxx_std_17)
target_link_libraries(StellaViewer.ClipFixtures PRIVATE clip_sqlite)
if(MSVC)
    target_compile_options(StellaViewer.ClipPreview PRIVATE /utf-8 /W4 /WX)
endif()
